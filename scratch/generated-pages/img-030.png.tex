\textbf{Finalmente calculamos el nivel de exposición a 2 m.}
\begin{description}
    \item[Datos] 
    \begin{itemize}
        \item $\lambda_1 = 1.03 \ \text{R/h}$
        \item $d_1 = 1 \ \text{m}$
        \item $d_2 = 2 \ \text{m}$
    \end{itemize}
    \item[Fórmula] 
    \[
    \lambda_2 = \left(\frac{d_1}{d_2}\right)^2 \times \lambda_1
    \]
    \[
    \lambda_2 = \left(\frac{1}{2}\right)^2 \times (1.03) = 0.257 \ \text{R/h}
    \]
    \item [\textbf{Respuesta}] $\lambda_2 = ?$
    \[
    \lambda_2 = 257 \ \text{mR/h}
    \]
\end{description}
