% Diagrama de Operación Radiográfica
\section*{Operación Radiográfica}

% Descripción de posiciones de la cámara y mangueras de guía
\subsection*{Posición Segura (Stored Position)}
\begin{itemize}
    \item \textbf{Telemando (Remote Control)}
    \item \textbf{Cámara (Projector)}: La cámara está situada a una distancia de 14.3 metros del telemando.
    \item \textbf{Manguera de Salida (Guide Tubes)}: La manguera tiene una longitud de 6 metros conectando con una manguera adicional de 2 metros, sumando un total de 8 metros hasta el telemando.
\end{itemize}

% Movimiento hacia la posición de exposición
\subsection*{Movimiento hacia la Posición de Exposición (In Transit to the Focal Point)}
\begin{itemize}
    \item La cámara se mueve desde la posición segura hasta el punto focal.
    \item El recorrido involucra el desplazamiento del telemando y la cámara a través de las mangueras guía mencionadas.
\end{itemize}

% Movimiento hacia la posición segura
\subsection*{Movimiento Hacia la Posición Segura (In Transit to the Stored Position)}
\begin{itemize}
    \item Después de la operación, la cámara y el resto del sistema regresan a la posición segura.
    \item Este movimiento es inverso al descrito en la sección de movimiento hacia la posición de exposición.
\end{itemize}
